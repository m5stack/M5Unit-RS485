71// Shared body of the two addAtomicBaseUART overloads; cfg is an Arduino SERIAL_* value.
72inline bool add_atomic_base_uart(UnitUnified& units, Component& unit, const uint32_t baud, const uint32_t cfg,
73 const size_t rx_buffer_size)
74{
75 const auto p = atomicBaseUartPins();
76 if (p.rx < 0 || p.tx < 0) {
77 M5_LIB_LOGE("wiring: addAtomicBaseUART unsupported board=0x%02x", static_cast<int>(M5.getBoard()));
78 return false;
79 }
80 M5_LIB_LOGI("wiring: addAtomicBaseUART board=0x%02x rx=%d tx=%d baud=%lu", static_cast<int>(M5.getBoard()),
81 static_cast<int>(p.rx), static_cast<int>(p.tx), static_cast<unsigned long>(baud));
82
83 HardwareSerial& serial = m5::unit::wiring::defaultUartSerial();
84 return m5::unit::wiring::beginUartSerial(serial, baud, cfg, p.rx, p.tx, rx_buffer_size) && units.add(unit, serial);
85}
