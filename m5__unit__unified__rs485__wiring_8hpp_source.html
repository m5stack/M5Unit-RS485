78inline bool addAtomicBaseUART(UnitUnified& units, Component& unit, const uint32_t baud = 115200,

115 auto port = m5::unit::wiring::uartPortHandle(m5::unit::wiring::defaultUartPort(), static_cast<gpio_num_t>(p.rx),
116 static_cast<gpio_num_t>(p.tx), baud, config);
117 if (port == UART_NUM_MAX) {

119 }
